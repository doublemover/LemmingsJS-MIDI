cmake_minimum_required(VERSION 3.24)
project(lemmings_native_midi_companion LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(FetchContent)
find_package(Git REQUIRED)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../midi-codec/dependencies.json" dependencies)
string(JSON repository GET "${dependencies}" libremidi repository)
string(JSON revision GET "${dependencies}" libremidi revision)
string(LENGTH "${revision}" revision_length)
if(NOT revision_length EQUAL 40 OR NOT revision MATCHES "^[0-9a-f]+$")
  message(FATAL_ERROR "An immutable libremidi revision is required")
endif()
foreach(option IN ITEMS EXAMPLES TESTS FIND_BOOST NI_MIDI2 DOWNLOAD_CPPWINRT)
  set(LIBREMIDI_${option} OFF CACHE BOOL "" FORCE)
endforeach()
foreach(option IN ITEMS NO_BOOST NO_JACK NO_PIPEWIRE NO_NETWORK NO_KEYBOARD NO_WINMM NO_WINUWP NO_KDMAPI)
  set(LIBREMIDI_${option} ON CACHE BOOL "" FORCE)
endforeach()
add_compile_definitions(LIBREMIDI_NO_BOOST_VARIANT2=1)
if(WIN32)
  add_compile_definitions(NOMINMAX=1 WIN32_LEAN_AND_MEAN=1)
  file(READ "${CMAKE_CURRENT_SOURCE_DIR}/sdk-dependencies.json" sdk_dependencies)
  string(JSON sdk_hash GET "${sdk_dependencies}" windowsMidiServices sha256)
  if(NOT EXISTS "${LIBREMIDI_WINMIDI_HEADERS_ZIP}")
    message(FATAL_ERROR "Run npm run build:native-midi to supply the pinned Windows MIDI SDK")
  endif()
  file(SHA256 "${LIBREMIDI_WINMIDI_HEADERS_ZIP}" actual_sdk_hash)
  if(NOT actual_sdk_hash STREQUAL sdk_hash)
    message(FATAL_ERROR "Windows MIDI SDK archive differs from the immutable lock")
  endif()
  get_filename_component(windows_kits "[HKEY_LOCAL_MACHINE\\SOFTWARE\\Microsoft\\Windows Kits\\Installed Roots;KitsRoot10]" ABSOLUTE)
  file(GLOB sdk_versions LIST_DIRECTORIES TRUE "${windows_kits}/Include/10.*")
  list(SORT sdk_versions COMPARE NATURAL ORDER DESCENDING)
  foreach(sdk_version IN LISTS sdk_versions)
    if(EXISTS "${sdk_version}/cppwinrt/winrt/base.h")
      get_filename_component(sdk_number "${sdk_version}" NAME)
      set(CPPWINRT_PATH "${sdk_version}/cppwinrt" CACHE PATH "Installed Windows SDK headers")
      set(WINRT_HEADER_PATH "${CPPWINRT_PATH}" CACHE PATH "Installed WinRT headers")
      find_program(CPPWINRT_TOOL NAMES cppwinrt.exe PATHS "${windows_kits}/bin/${sdk_number}/x64" REQUIRED NO_DEFAULT_PATH)
      set(CMAKE_VS_WINDOWS_TARGET_PLATFORM_VERSION "${sdk_number}")
      break()
    endif()
  endforeach()
  if(NOT CPPWINRT_PATH OR NOT CPPWINRT_TOOL)
    message(FATAL_ERROR "Install a Windows SDK with cppwinrt; unversioned downloads are disabled")
  endif()
  include_directories(SYSTEM "${CPPWINRT_PATH}")
endif()
FetchContent_Declare(libremidi GIT_REPOSITORY "${repository}" GIT_TAG "${revision}" GIT_SUBMODULES "" GIT_CONFIG core.autocrlf=false)
FetchContent_MakeAvailable(libremidi)
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${libremidi_SOURCE_DIR}" rev-parse HEAD OUTPUT_VARIABLE actual OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT actual STREQUAL revision)
  message(FATAL_ERROR "libremidi checkout differs from the lock")
endif()
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${libremidi_SOURCE_DIR}" diff --quiet HEAD RESULT_VARIABLE dirty)
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${libremidi_SOURCE_DIR}" ls-files --others --exclude-standard OUTPUT_VARIABLE untracked OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT dirty EQUAL 0 OR NOT untracked STREQUAL "")
  message(FATAL_ERROR "The pinned dependency source was modified")
endif()
string(JSON ni_repository GET "${dependencies}" ni_midi2 repository)
string(JSON ni_revision GET "${dependencies}" ni_midi2 revision)
string(LENGTH "${ni_revision}" ni_revision_length)
if(NOT ni_revision_length EQUAL 40 OR NOT ni_revision MATCHES "^[0-9a-f]+$")
  message(FATAL_ERROR "An immutable ni-midi2 revision is required")
endif()
set(NIMIDI2_TESTS OFF CACHE BOOL "" FORCE)
set(NIMIDI2_EXAMPLES OFF CACHE BOOL "" FORCE)
FetchContent_Declare(ni_midi2 GIT_REPOSITORY "${ni_repository}" GIT_TAG "${ni_revision}" GIT_SUBMODULES "" GIT_CONFIG core.autocrlf=false)
FetchContent_MakeAvailable(ni_midi2)
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${ni_midi2_SOURCE_DIR}" rev-parse HEAD OUTPUT_VARIABLE ni_actual OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${ni_midi2_SOURCE_DIR}" diff --quiet HEAD RESULT_VARIABLE ni_dirty)
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${ni_midi2_SOURCE_DIR}" ls-files --others --exclude-standard OUTPUT_VARIABLE ni_untracked OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT ni_actual STREQUAL ni_revision OR NOT ni_dirty EQUAL 0 OR NOT ni_untracked STREQUAL "")
  message(FATAL_ERROR "ni-midi2 source differs from its immutable lock")
endif()
add_executable(lemmings-midi-companion companion.cpp)
target_link_libraries(lemmings-midi-companion PRIVATE libremidi ni::midi2)
target_compile_definitions(lemmings-midi-companion PRIVATE LIBREMIDI_REVISION="${revision}" NI_MIDI2_REVISION="${ni_revision}")
if(CMAKE_CXX_COMPILER_FRONTEND_VARIANT STREQUAL "MSVC")
  target_compile_options(lemmings-midi-companion PRIVATE /W4 /EHsc /bigobj)
else()
  target_compile_options(lemmings-midi-companion PRIVATE -Wall -Wextra -Wpedantic)
endif()
install(TARGETS lemmings-midi-companion RUNTIME DESTINATION bin)
